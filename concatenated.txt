#[derive(Clone, Debug)]
pub struct ParametersFormat<'a> {
    parameters: &'a Parameters,
    parameter_formats: Vec<ParameterFormat>,
}

impl<'a> ParametersFormat<'a> {
    pub fn new(parameters: &'a Parameters) -> Self {
        Self {
            parameters,
            parameter_formats: Vec::new(),
        }
    }

    pub fn push(&mut self, parameter_format: ParameterFormat) {
        self.parameter_formats.push(parameter_format);
    }
}

impl Display for ParametersFormat<'_> {
    fn fmt(&self, f: &mut Formatter) -> Result {
        for parameter_format in &self.parameter_formats {
            // Создаем итератор, который оставляет только подходящие под фильтр параметры
            let mut matched_parameters = self
                .parameters
                .iter()
                .filter(|parameter| parameter_format.rule.matches(&parameter.name));
            match parameter_format.value {
                Value::All => {
                    for parameter in matched_parameters {
                        write!(f, "{{{parameter}}}")?;
                    }
                }
                Value::First => {
                    if let Some(parameter) = matched_parameters.next() {
                        write!(f, "{{{parameter}}}")?;
                    }
                }
                Value::Last => {
                    if let Some(parameter) = matched_parameters.last() {
                        write!(f, "{{{parameter}}}")?;
                    }
                }
            }
        }
        Ok(())
    }
}

impl Writable<'_> {
    pub fn show(&mut self, ui: &mut Ui) {
        let mut sort_clicked = false;
        let mut to_remove = Vec::new();

        let header_height = ui.text_style_height(&TextStyle::Heading);
        let body_height = ui.text_style_height(&TextStyle::Body);
        let width = ui.style().spacing.text_edit_width;
        // Отрисовываем таблицу параметров
        TableBuilder::new(ui)
            .striped(true) // Полосатый фон для лучшей читаемости строк
            .resizable(true)
            .cell_layout(egui::Layout::left_to_right(egui::Align::Center))
            .column(Column::auto().resizable(false)) // Кнопка удаления (MINUS)
            .column(Column::auto_with_initial_suggestion(width)) // Поле ключа (Name)
            .column(Column::remainder()) // Поле значения (Value)
            // .column(Column::auto_with_initial_suggestion(width)) // Поле значения (Value)
            .header(header_height, |mut header| {
                header.col(|ui| {
                    ui.add_enabled_ui(!self.parameters.is_empty(), |ui| {
                        if ui.button(SORT_ASCENDING).clicked() {
                            sort_clicked = true;
                        }
                    });
                });
                header.col(|ui| {
                    ui.heading("Name");
                });
                header.col(|ui| {
                    ui.heading("Value");
                });
            })
            .body(|body| {
                body.rows(body_height, self.parameters.len(), |mut row| {
                    let index = row.index();
                    let parameter = &mut self.parameters[index];

                    // Столбец 1: Кнопка удаления
                    row.col(|ui| {
                        if ui.button(MINUS).clicked() {
                            to_remove.push(index);
                        }
                    });

                    // Столбец 2: Поле ключа (имени)
                    row.col(|ui| {
                        let response = TextEdit::singleline(&mut parameter.name)
                            .desired_width(f32::INFINITY) // Заполняем всю ширину колонки
                            .ui(ui);
                        if response.lost_focus() || response.clicked_elsewhere() {
                            parameter.name = parameter.name.trim().to_owned();
                        }
                    });

                    // Столбец 4: Поле значения со специальными эдиторами
                    row.col(|ui| {
                        match &*parameter.name {
                            DATE => {
                                let mut date = parameter.value.parse().unwrap_or_default();
                                if DatePickerButton::new(&mut date)
                                    .id_salt(&ui.next_auto_id().value().to_string())
                                    .ui(ui)
                                    .changed()
                                {
                                    parameter.value = date.to_string();
                                }
                            }
                            IDENTIFIER => {
                                let mut number = parameter.value.parse::<u64>().unwrap_or(0);
                                if DragValue::new(&mut number).speed(0.1).ui(ui).changed() {
                                    parameter.value = number.to_string();
                                }
                            }
                            DESCRIPTION => {
                                // Многострочный редактор для описания
                                let response = TextEdit::multiline(&mut parameter.value)
                                    // .desired_width(f32::INFINITY)
                                    .ui(ui);
                                if response.lost_focus() || response.clicked_elsewhere() {
                                    parameter.value = parameter.value.trim().to_owned();
                                }
                            }
                            _ => {
                                // Стандартный однострочный редактор для всех остальных
                                let response = TextEdit::singleline(&mut parameter.value)
                                    // .desired_width(f32::INFINITY)
                                    .ui(ui);
                                if response.lost_focus() || response.clicked_elsewhere() {
                                    parameter.value = parameter.value.trim().to_owned();
                                }
                            }
                        }
                    });
                });
            });

        // Обработка отложенных действий (удаление и сортировка)
        // Выполняем после отрисовки таблицы, чтобы избежать проблем с заимствованием (borrow checker)

        if !to_remove.is_empty() {
            let mut i = 0;
            self.parameters.retain_mut(|_| {
                let keep = !to_remove.contains(&i);
                i += 1;
                keep
            });
        }

        if sort_clicked {
            self.parameters
                .retain_mut(|parameter| !parameter.name.is_empty() || !parameter.value.is_empty());
            self.parameters.filter_and_sort([&*ALL_ALL_ASCENDING]);
        }

        // Кнопка добавления
        if ui.button(PLUS).clicked() {
            self.parameters.push(Parameter::default());
        }
    }
}

pub fn join_with<'a, T: Borrow<Rule>>(
    iter: impl Iterator<Item = &'a Metadata> + Clone,
    rules: impl AsRef<[T]>,
) -> Metadata {
    let parameters = iter.flat_map(|metadata| &metadata.0).cloned().collect();

    metadata.filter_and_sort(rules);
    metadata
}

/// Metadata
#[derive(Clone, Debug, Default, Deserialize, Eq, Hash, Ord, PartialEq, PartialOrd, Serialize)]
pub struct Metadata(Vec<Parameter>);

impl Metadata {
    pub fn new() -> Self {
        Self(Vec::new())
    }

    pub fn push_name_value(&mut self, name: impl Into<String>, value: impl ToString) {
        self.0.push(Parameter {
            name: name.into(),
            value: value.to_string(),
        });
    }

    pub fn filter_name(&self, name: &str) -> impl Iterator<Item = &str> {
        self.0
            .iter()
            .filter_map(move |parameter| (parameter.name == name).then_some(&*parameter.value))
    }

    pub fn find_name(&self, name: &str) -> Option<&str> {
        self.0
            .iter()
            .find_map(|parameter| (parameter.name == name).then_some(&*parameter.value))
    }

    /// Значения всех параметров с данным именем в порядке следования.
    fn values<'a>(&'a self, name: &'a str) -> impl Iterator<Item = &'a String> {
        self.0
            .iter()
            .filter(move |parameter| parameter.name == name)
            .map(|parameter| &parameter.value)
    }

    /// Сравнивает две метаданные по списку полей: сначала по первому полю,
    /// при равенстве по следующему и т. д. Если поля нет, значений ноль,
    /// и такая запись идёт раньше (при `Descending`, наоборот, позже).
    pub fn cmp_by<S: AsRef<str>>(&self, other: &Self, sort: &[(S, Order)]) -> std::cmp::Ordering {
        for (name, order) in sort {
            let name = name.as_ref();
            let ordering = self.values(name).cmp(other.values(name));
            let ordering = match order {
                Order::Ascending => ordering,
                Order::Descending => ordering.reverse(),
            };
            if ordering.is_ne() {
                return ordering;
            }
        }
        std::cmp::Ordering::Equal
    }

    pub fn contains_name_value(&self, name: &str, value: &str) -> bool {
        self.0
            .iter()
            .find(|parameter| parameter.name == name && parameter.value == value)
            .is_some()
    }

    /// Оставляет только те параметры, которые попали хотя бы под одно правило,
    /// фильтрует дубликаты согласно `Rule::value` и выстраивает их в порядке правил.
    pub fn filter_and_sort<T: Borrow<Rule>>(&mut self, rules: impl AsRef<[T]>) {
        let rules = rules.as_ref();
        let parameters = std::mem::take(&mut self.0);

        // Фильтруем по имени
        // Находим первое подходящее правило для каждого параметра
        let rank_value_sort_parameters: Vec<_> = parameters
            .into_iter()
            .filter_map(|parameter| {
                // Ищем подходящее правило и захватываем его политику сортировки
                let (rank, value, sort) = rules.iter().enumerate().find_map(|(rank, rule)| {
                    let rule = rule.borrow();
                    if rule.name.matches(&parameter.name) {
                        Some((rank, rule.value, rule.sort))
                    } else {
                        None
                    }
                })?;
                Some((rank, value, sort, parameter))
            })
            .collect();

        // Применяем политики First, Last и Unique
        let mut keep = vec![true; rank_value_sort_parameters.len()];
        let mut first = HashSet::new();
        let mut last = HashSet::new();
        let mut unique = HashSet::new();

        // Обрабатываем First и Unique (идем с начала)
        for (index, (_, value, _, parameter)) in rank_value_sort_parameters.iter().enumerate() {
            match value {
                Value::First => {
                    if !first.insert(&parameter.name) {
                        keep[index] = false;
                    }
                }
                Value::Unique => {
                    if !unique.insert((&parameter.name, &parameter.value)) {
                        keep[index] = false;
                    }
                }
                _ => {}
            }
        }

        // Обрабатываем Last (идем с конца)
        for (index, (_, value, _, parameter)) in rank_value_sort_parameters.iter().enumerate().rev()
        {
            if let Value::Last = value {
                if !last.insert(&parameter.name) {
                    keep[index] = false;
                }
            }
        }

        // Фильтруем по значению
        let mut rank_parameters: Vec<_> = rank_value_sort_parameters
            .into_iter()
            .enumerate()
            .filter_map(|(index, (rank, _, sort, parameter))| {
                if keep[index] {
                    Some((rank, sort, parameter))
                } else {
                    None
                }
            })
            .collect();

        // Сортируем
        // rank_parameters.sort_by_cached_key(|(rank, sort, parameter)| {
        //     (*rank, *sort, *parameter)
        //     // left_rank
        //     //     .cmp(right_rank)
        //     //     // Если параметры попали под одно и то же правило (ранги равны)
        //     //     .then_with(|| {
        //     //         // Поскольку left_rank == right_rank, то и left_sort == right_sort
        //     //         match left_sort {
        //     //             Some(Order::Ascending) => left_parameter
        //     //                 .name
        //     //                 .cmp(&right_parameter.name)
        //     //                 .then_with(|| left_parameter.value.cmp(&right_parameter.value)),
        //     //             // Возвращаем Equal, чтобы стабильная сортировка сохранила исходный порядок
        //     //             None => std::cmp::Ordering::Equal,
        //     //         }
        //     //     })
        // });
        rank_parameters.sort_by(
            |(left_rank, left_sort, left_parameter), (right_rank, _right_sort, right_parameter)| {
                left_rank
                    .cmp(right_rank)
                    // Если параметры попали под одно и то же правило (ранги равны)
                    .then_with(|| {
                        // Поскольку left_rank == right_rank, то и left_sort == right_sort
                        match left_sort {
                            Some(Order::Ascending) => left_parameter
                                .name
                                .cmp(&right_parameter.name)
                                .then_with(|| left_parameter.value.cmp(&right_parameter.value)),
                            Some(Order::Descending) => right_parameter
                                .name
                                .cmp(&left_parameter.name)
                                .then_with(|| right_parameter.value.cmp(&left_parameter.value)),
                            // Возвращаем Equal, чтобы стабильная сортировка сохранила исходный порядок
                            None => std::cmp::Ordering::Equal,
                        }
                    })
            },
        );

        // Убираем ранг и возвращаем обратно в self
        self.0 = rank_parameters
            .into_iter()
            .map(|(_, _, parameter)| parameter)
            .collect();
    }

    pub fn format(&self) -> impl Display {
        let mut parameters = self.clone();
        parameters.filter_and_sort([&*ALL_NAMES, &*ALL_IDENTIFIERS, &*LAST_DATE, &*ALL_VERSIONS]);
        from_fn(move |f| write!(f, "{parameters}"))
    }

    pub fn format_with<T: Borrow<Rule>>(&self, rules: impl AsRef<[T]>) -> impl Display {
        let mut parameters = self.clone();
        parameters.filter_and_sort(rules);
        from_fn(move |f| write!(f, "{parameters}"))
    }
}

            value: String::new(),
        }
    }

        Ok(())
    }
}

/// Version
#[derive(
    Clone, Copy, Debug, Default, Deserialize, Eq, Hash, Ord, PartialEq, PartialOrd, Serialize,
)]
pub struct Version(pub u64, pub u64, pub u64);

impl Version {
    fn new() -> Self {
        Self(0, 0, 0)
    }
}

impl Display for Version {
    fn fmt(&self, f: &mut Formatter) -> std::fmt::Result {
        if self.0 != 0 {
            write!(f, "{}", self.0)?;
        }
        if self.1 != 0 {
            if self.0 != 0 {
                f.write_str(".")?;
            }
            write!(f, "{}", self.1)?;
        }
        if self.0 != 0 || self.1 != 0 {
            f.write_str(".")?;
        }
        write!(f, "{}", self.2)?;
        Ok(())
    }
}

/// Order

/// Rule

/// Name
#[derive(Clone, Debug, Eq, Ord, PartialEq, PartialOrd)]
pub enum Name {
    /// Точное совпадение имени параметра
    Exact(String),
    /// Логическое NOT - всё, кроме указанных в списке
    Not(Box<Self>),
    /// Логическое ИЛИ для сложных вложенных правил
    Or(Vec<Name>),
    /// Логическое И для сложных вложенных правил
    And(Vec<Name>),
    /// Захватить вообще всё
    All,
}

impl Name {
    pub fn exact(name: impl Into<String>) -> Self {
        Name::Exact(name.into())
    }

    pub fn not(expr: Name) -> Self {
        Name::Not(Box::new(expr))
    }

    pub fn or(exprs: impl IntoIterator<Item = Name>) -> Self {
        Name::Or(exprs.into_iter().collect())
    }

    pub fn and(exprs: impl IntoIterator<Item = Name>) -> Self {
        Name::And(exprs.into_iter().collect())
    }
}

impl Name {
    /// Проверяет, подходит ли имя параметра под данное выражение
    pub fn matches(&self, target: &str) -> bool {
        match self {
            Name::Exact(name) => target == name,
            Name::Not(name) => !name.matches(target),
            Name::Or(names) => names.iter().any(|name| name.matches(target)),
            Name::And(names) => names.iter().all(|name| name.matches(target)),
            Name::All => true,
        }
    }
}

#[derive(Clone, Copy, Debug, Default, Eq, Ord, PartialEq, PartialOrd)]

    #[default]
    All,
    First,
    Last,
    Unique,
}
